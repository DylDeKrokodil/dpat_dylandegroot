package "Domain model" {
    class FsmDiagram {
        - states: Dictionary of State
        - triggers: Dictionary of Trigger
        - transitions: List of Transition
        --
        + States: ReadOnlyCollection of State
        + Triggers: ReadOnlyCollection of Trigger
        + Transitions: ReadOnlyCollection of Transition
        + InitialState: InitialState or null
        --
        + AddState(state: State): void
        + AddTrigger(trigger: Trigger): void
        + AddTransition(transition: Transition): void
        + FindState(id: string): State or null
        + FindTrigger(id: string): Trigger or null
        + GetOutgoingTransitions(state: State): Collection of Transition
        + GetIncomingTransitions(state: State): Collection of Transition
        + Accept(visitor: IFsmElementVisitor): void
    }

    abstract class State {
        + Id: string
        + DisplayName: string
        + Parent: State or null
        + Actions: ReadOnlyCollection of StateAction
        --
        + AddAction(action: StateAction): void
        + IsNestedIn(possibleParent: State): bool
        + Accept(visitor: IFsmElementVisitor): void
    }

    class InitialState {
        + InitialState(id: string, displayName: string, parent: State or null)
    }

    class SimpleState {
        + SimpleState(id: string, displayName: string, parent: State or null)
    }

    class CompoundState {
        - children: List of State
        --
        + Children: ReadOnlyCollection of State
        + CompoundState(id: string, displayName: string, parent: State or null)
        + AddChild(child: State): void
    }

    class FinalState {
        + FinalState(id: string, displayName: string, parent: State or null)
    }

    class Transition {
        + Id: string
        + Source: State
        + Destination: State
        + Trigger: Trigger or null
        + Guard: Guard
        + Effect: TransitionAction or null
        --
        + IsAutomatic(): bool
        + IsSelfTransition(): bool
        + SetEffect(action: TransitionAction): void
        + Accept(visitor: IFsmElementVisitor): void
    }

    class Trigger {
        + Id: string
        + Description: string
    }

    class Guard {
        + Expression: string
        --
        + IsEmpty(): bool
    }

    abstract class FsmAction {
        + Description: string
        + Type: ActionType
    }

    class StateAction {
        + Owner: State
    }

    class TransitionAction {
        + Owner: Transition
    }

    enum ActionType {
        EntryAction
        DoAction
        ExitAction
        TransitionAction
    }

    interface IFsmElementVisitor {
        + VisitDiagram(diagram: FsmDiagram): void
        + VisitInitialState(state: InitialState): void
        + VisitSimpleState(state: SimpleState): void
        + VisitCompoundState(state: CompoundState): void
        + VisitFinalState(state: FinalState): void
        + VisitTransition(transition: Transition): void
    }
}

package "Building and factory" {
    class FsmModelBuilder {
        - diagram: FsmDiagram
        --
        + Diagram: FsmDiagram
        --
        + AddState(id: string, parentId: string, name: string, type: StateType): State
        + AddTrigger(id: string, description: string): Trigger
        + AddTransition(id: string, sourceId: string, destinationId: string, triggerId: string or null, guard: string): Transition
        + AddAction(ownerId: string, description: string, type: ActionType): FsmAction
        + Build(): FsmDiagram
    }

    class StateFactory {
        + Create(id: string, name: string, type: StateType, parent: State or null): State
    }

    enum StateType {
        Initial
        Simple
        Compound
        Final
    }
}

package "Parsing" {
    interface IFsmParser {
        + ParseFile(filePath: string): FsmDiagram
        + ParseText(text: string): FsmDiagram
    }

    class FsmTextParser {
        - tokenizer: FsmTokenizer
        - builder: FsmModelBuilder
        --
        + ParseFile(filePath: string): FsmDiagram
        + ParseText(text: string): FsmDiagram
    }

    class FsmTokenizer {
        + Tokenize(text: string): Collection of FsmDefinition
    }

    class FsmDefinition {
        + Type: DefinitionType
        + RawText: string
        + LineNumber: int
    }

    enum DefinitionType {
        State
        Trigger
        Action
        Transition
    }

    class ParseException {
        + LineNumber: int
        + Message: string
    }
}

package "Validation" {
    interface IFsmValidator {
        + Validate(diagram: FsmDiagram): Collection of ValidationError
    }

    class ValidationPipeline {
        - validators: Collection of IFsmValidator
        --
        + Validate(diagram: FsmDiagram): ValidationResult
    }

    class ValidationResult {
        + Errors: Collection of ValidationError
        + IsValid: bool
        --
        + Success(): ValidationResult
        + Failed(errors: Collection of ValidationError): ValidationResult
    }

    class ValidationError {
        + Code: string
        + Message: string
        + ElementId: string or null
    }

    class DeterministicTransitionValidator {
        + Validate(diagram: FsmDiagram): Collection of ValidationError
    }

    class InitialFinalTransitionValidator {
        + Validate(diagram: FsmDiagram): Collection of ValidationError
    }

    class UnreachableStateValidator {
        + Validate(diagram: FsmDiagram): Collection of ValidationError
    }
}

package "Presentation" {
    interface IFsmRenderer {
        + RenderDiagram(diagram: FsmDiagram): string
        + RenderState(state: State): string
        + RenderTransition(transition: Transition): string
    }

    class ConsoleTextRenderer {
        - builder: TextOutputBuilder
        - context: RenderContext
        --
        + RenderDiagram(diagram: FsmDiagram): string
        + RenderState(state: State): string
        + RenderTransition(transition: Transition): string
        + VisitDiagram(diagram: FsmDiagram): void
        + VisitInitialState(state: InitialState): void
        + VisitSimpleState(state: SimpleState): void
        + VisitCompoundState(state: CompoundState): void
        + VisitFinalState(state: FinalState): void
        + VisitTransition(transition: Transition): void
    }

    class RenderContext {
        + IndentationLevel: int
        --
        + Indent(): string
        + NextLevel(): RenderContext
    }

    class TextOutputBuilder {
        + AppendLine(text: string): void
        + ToString(): string
        + Clear(): void
    }
}

package "Application and UI" {
    class FsmApplication {
        - parser: IFsmParser
        - validationPipeline: ValidationPipeline
        - renderer: IFsmRenderer
        - userInterface: IUserInterface
        --
        + Run(args: StringArray): int
    }

    interface IUserInterface {
        + GetInputFilePath(args: StringArray): string or null
        + ShowOutput(output: string): void
        + ShowErrors(errors: Collection of string): void
    }

    class ConsoleUserInterface {
        + GetInputFilePath(args: StringArray): string or null
        + ShowOutput(output: string): void
        + ShowErrors(errors: Collection of string): void
    }
}
